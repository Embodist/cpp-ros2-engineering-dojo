cmake_minimum_required(VERSION 3.20)
project(cpp_ros2_engineering_dojo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(DOJO_SANITIZE_ADDRESS "Enable ASan/UBSan (GCC/Clang only)" OFF)
option(DOJO_SANITIZE_THREAD "Enable TSan (GCC/Clang only)" OFF)
option(WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
option(ENABLE_CLANG_TIDY "Run clang-tidy during build" OFF)
option(DOJO_USE_GTEST "Build the optional GoogleTest mirror suite" OFF)
option(DOJO_BUILD_PROJECTS "Build the ported cpp_testing reference projects" OFF)
set(DOJO_GTEST_SOURCE_DIR "" CACHE PATH "Local GoogleTest source for offline builds")
if(DOJO_SANITIZE_ADDRESS AND DOJO_SANITIZE_THREAD)
  message(FATAL_ERROR "ASan and TSan cannot be used together")
endif()
add_library(dojo_warnings INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dojo_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(WARNINGS_AS_ERRORS)
    target_compile_options(dojo_warnings INTERFACE -Werror)
  endif()
  if(DOJO_SANITIZE_ADDRESS)
    target_compile_options(dojo_warnings INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dojo_warnings INTERFACE -fsanitize=address,undefined)
  elseif(DOJO_SANITIZE_THREAD)
    target_compile_options(dojo_warnings INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(dojo_warnings INTERFACE -fsanitize=thread)
  endif()
endif()

if(ENABLE_CLANG_TIDY)
  find_program(CLANG_TIDY_EXE NAMES clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_EXE}")
endif()

if(DOJO_USE_GTEST)
  if(DOJO_GTEST_SOURCE_DIR)
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${DOJO_GTEST_SOURCE_DIR}"
        CACHE PATH "Local GoogleTest source" FORCE)
  endif()
  include(FetchContent)
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.15.2
  )
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  include(GoogleTest)
endif()

include(CTest)
add_subdirectory(stage1)
if(DOJO_BUILD_PROJECTS)
  add_subdirectory(stage1/projects)
endif()
